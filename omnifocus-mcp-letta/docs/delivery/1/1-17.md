#### **Duration Filters**
```typescript
minDuration: number;   // Minimum estimated duration in minutes
```

#### **Basic Status Filters**
```typescript
flagged: boolean;      // Filter for flagged/unflagged tasks

blocked: boolean;      // Filter for blocked/unblocked tasks
dropped: boolean;      // Filter for dropped/active tasks (distinct from includeDropped)
```

## Implementation Strategy

### **Step 1: Add Date Parsing Logic**
Add the same date pre-parsing logic that `queryTasks` uses:

```javascript
// Pre-parse date filters for efficiency (add to searchTasks)
const dueBefore = params.dueBefore ? new Date(params.dueBefore) : null;
const dueAfter = params.dueAfter ? new Date(params.dueAfter) : null;
const deferBefore = params.deferBefore ? new Date(params.deferBefore) : null;
const deferAfter = params.deferAfter ? new Date(params.deferAfter) : null;
```

### **Step 2: Add Missing Filters**
Insert the missing filter logic in the existing metadata filtering section:

```javascript
// Add to existing searchTasks metadata filtering
searchTasks = searchTasks.filter(task => {
  // ... existing advanced metadata filters ...
  
  // Date range filters - NEW for Task 1-17
  if (dueBefore && (!task.dueDate || task.dueDate >= dueBefore)) return false;
  if (dueAfter && (!task.dueDate || task.dueDate <= dueAfter)) return false;
  if (deferBefore && (!task.deferDate || task.deferDate >= deferBefore)) return false;
  if (deferAfter && (!task.deferDate || task.deferDate <= deferAfter)) return false;
  
  // Duration filters - NEW for Task 1-17
  if (params.minDuration !== undefined && (!task.estimatedMinutes || task.estimatedMinutes < params.minDuration)) return false;
  if (params.maxDuration !== undefined && (!task.estimatedMinutes || task.estimatedMinutes > params.maxDuration)) return false;
  
  // Basic status filters - NEW for Task 1-17
  if (params.flagged !== undefined && task.flagged !== params.flagged) return false;
  if (params.completed !== undefined && task.completed !== params.completed) return false;
  if (params.dropped !== undefined && !task.effectiveActive !== params.dropped) return false;
  if (params.blocked !== undefined) {
    const isBlocked = task.taskStatus === Task.Status.Blocked;
    if (params.blocked !== isBlocked) return false;
  }
  
  return true;
});
```
