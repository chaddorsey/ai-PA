# Amtrak Companion — Plan 2: Headless Companion-Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Build `packages/companion-core/` — a pure TypeScript package, fully unit-testable under Node/vitest, that loads per-leg bundles and exposes the position projection, position service, scheduler, ETA, dive-grounding, and favorites APIs consumed by Plans 3 and 4.

**Architecture:** The package has no UI and no native dependencies; all platform-specific capabilities (GPS, audio, storage) are accessed through injectable adapter interfaces so the same code runs in Node tests, a browser, and a Capacitor shell without modification. The projection math is a direct port of the Python engine's `_milepost_latlon` and `project_to_leg` functions, golden-tested against fixture output generated by running the actual Python engine. The scheduler is the centerpiece and is tested with a simulated-train harness that feeds a synthetic position track and asserts the exact unit firing sequence, fill ratio, and filter behavior.

**Tech Stack:** TypeScript 5.x (strict), vitest 1.x, Node 20+, zero runtime dependencies outside stdlib types.

## Global Constraints
- Package location: `packages/companion-core/` inside the monorepo root.
- `"strict": true` in tsconfig; no `any` without an explicit cast comment.
- Zero native or UI dependencies — no Capacitor, no DOM, no file system imports in src/.
- The bundle format (`bundle.json`) and the companion-core public API (exact names and types listed below) are a locked contract — Plans 3 and 4 depend on them verbatim.
- Projection must be golden-tested: fixture file generated by running Python `position_engine.py`, TS result must match within 0.001° lat/lon and 0.01 mi.
- Tests use small synthetic fixtures plus the proxy bundle produced by Plan 1 (Task 7 of that plan creates the position table; the proxy bundle is the CONO leg-58 sampler or a synthetic stand-in until Plan 1 ships).
- No commits that skip a failing test phase — every task follows the red → green → commit sequence.
- One commit per task, message format: `feat(companion-core): <task-name>`.

import type { Bundle, Unit, SquibUnit, InterstitialUnit } from './types.js';

function assertString(v: unknown, field: string): string {
  if (typeof v !== 'string' || v.length === 0)
    throw new Error(`Bundle validation: "${field}" must be a non-empty string`);
  return v;
}

function assertNumber(v: unknown, field: string): number {
  if (typeof v !== 'number' || !isFinite(v))
    throw new Error(`Bundle validation: "${field}" must be a finite number`);
  return v;
}

function validateUnit(raw: unknown, idx: number): Unit {
  if (typeof raw !== 'object' || raw === null)
    throw new Error(`Bundle validation: units[${idx}] must be an object`);
  const u = raw as Record<string, unknown>;

  assertString(u['id'], `units[${idx}].id`);
  assertString(u['text'], `units[${idx}].text`);
  assertString(u['audio'], `units[${idx}].audio`);
  assertString(u['place'], `units[${idx}].place`);
  assertString(u['theme'], `units[${idx}].theme`);
  assertNumber(u['lat'], `units[${idx}].lat`);
  assertNumber(u['lon'], `units[${idx}].lon`);
  assertNumber(u['dur_s'], `units[${idx}].dur_s`);

  const salience = u['salience'];
  if (typeof salience !== 'number' || salience < 1 || salience > 5)
    throw new Error(`Bundle validation: units[${idx}].salience must be 1–5`);

  const kind = u['kind'];
  if (kind === 'squib') {
    assertNumber(u['mile'], `units[${idx}].mile`);
    return u as unknown as SquibUnit;
  } else if (kind === 'interstitial') {
    assertNumber(u['from_mi'], `units[${idx}].from_mi`);
    assertNumber(u['to_mi'], `units[${idx}].to_mi`);
    return u as unknown as InterstitialUnit;
  }
  throw new Error(`Bundle validation: units[${idx}].kind must be "squib" or "interstitial"`);
}

export function validateBundle(raw: unknown): Bundle {
  if (typeof raw !== 'object' || raw === null)
    throw new Error('Bundle validation: bundle must be an object');
  const b = raw as Record<string, unknown>;

  assertString(b['leg'], 'leg');

  if (!Array.isArray(b['units']))
    throw new Error('Bundle validation: "units" must be an array');
  const units: Unit[] = (b['units'] as unknown[]).map(validateUnit);

  const layers = b['layers'];
  if (typeof layers !== 'object' || layers === null)
    throw new Error('Bundle validation: "layers" must be an object');
  for (const key of ['guide', 'lore', 'science', 'connections', 'themes']) {
    if (!(key in (layers as object)))
      throw new Error(`Bundle validation: layers.${key} is missing`);
  }

  if (!Array.isArray(b['position_table']))
    throw new Error('Bundle validation: "position_table" must be an array');
  for (let i = 0; i < (b['position_table'] as unknown[]).length; i++) {
    const row = (b['position_table'] as unknown[])[i];
    if (!Array.isArray(row) || row.length !== 4)
      throw new Error(`Bundle validation: position_table[${i}] must be a 4-element array [elapsed_min, mile, lat, lon]`);
  }

  return {
    leg: b['leg'] as string,
    units,
    layers: layers as Bundle['layers'],
    position_table: b['position_table'] as Bundle['position_table'],
  };
}

/**
 * Load and validate a bundle JSON file. `resolvePath` is an injectable adapter
 * so this works in both Node (fs.readFile) and Capacitor (Filesystem plugin).
 */
export async function loadBundle(
  legId: string,
  resolvePath: (legId: string) => Promise<unknown>,
): Promise<Bundle> {
  const raw = await resolvePath(legId);
  return validateBundle(raw);
}

import type { Polyline, LatLon, ProjectionResult } from './types.js';

const EARTH_MI = 3958.7613;

function haversineMi(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const toR = Math.PI / 180;
  const dlat = (lat2 - lat1) * toR;
  const dlon = (lon2 - lon1) * toR;
  const h =
    Math.sin(dlat / 2) ** 2 +
    Math.cos(lat1 * toR) * Math.cos(lat2 * toR) * Math.sin(dlon / 2) ** 2;
  return 2 * EARTH_MI * Math.asin(Math.min(1.0, Math.sqrt(h)));
}

/**
 * Port of Python _milepost_latlon(poly, mile).
 * poly: [[anchor_mile, lat, lon], ...]
 */
export function milepostToLatLon(poly: Polyline, mile: number): LatLon {
  if (poly.length === 0) throw new Error('milepostToLatLon: empty polyline');
  if (mile <= poly[0][0]) return { lat: poly[0][1], lon: poly[0][2] };
  if (mile >= poly[poly.length - 1][0]) {
    const last = poly[poly.length - 1];
    return { lat: last[1], lon: last[2] };
  }
  for (let i = 0; i < poly.length - 1; i++) {
    const [m1, la1, lo1] = poly[i];
    const [m2, la2, lo2] = poly[i + 1];
    if (m1 <= mile && mile <= m2) {
      const f = m2 !== m1 ? (mile - m1) / (m2 - m1) : 0.0;
      return { lat: la1 + f * (la2 - la1), lon: lo1 + f * (lo2 - lo1) };
    }
  }
  const last = poly[poly.length - 1];
  return { lat: last[1], lon: last[2] };
}

/**
 * Project (qlat, qlon) onto one segment from (lat1,lon1)→(lat2,lon2).
 * Returns { t: [0,1], footLat, footLon }.
 * Port of Python _project_point_seg.
 */
function projectPointSeg(
  qlat: number, qlon: number,
  lat1: number, lon1: number,
  lat2: number, lon2: number,
): { t: number; footLat: number; footLon: number } {
  const kx = Math.cos(((lat1 + lat2) / 2) * (Math.PI / 180));
  const ax = (lon2 - lon1) * kx;
  const ay = lat2 - lat1;
  const bx = (qlon - lon1) * kx;
  const by = qlat - lat1;
  const denom = ax * ax + ay * ay;
  const t = denom === 0 ? 0.0 : Math.max(0.0, Math.min(1.0, (ax * bx + ay * by) / denom));
  return { t, footLat: lat1 + t * (lat2 - lat1), footLon: lon1 + t * (lon2 - lon1) };
}

/**
 * Port of Python project_to_leg(poly, lat, lon).
 * Returns { mile, offtrackMi, side: 'L' | 'R' }.
 */
export function projectToLeg(poly: Polyline, lat: number, lon: number): ProjectionResult {
  let bestDist = Infinity;
  let bestMile = 0;
  let bestSide: 'L' | 'R' = 'R';

  for (let i = 0; i < poly.length - 1; i++) {
    const [m1, la1, lo1] = poly[i];
    const [m2, la2, lo2] = poly[i + 1];
    const { t, footLat, footLon } = projectPointSeg(lat, lon, la1, lo1, la2, lo2);
    const d = haversineMi(lat, lon, footLat, footLon);
    if (d < bestDist) {
      bestDist = d;
      bestMile = m1 + t * (m2 - m1);
      // Cross product (travel direction) × (toward feature):
      // positive cross = feature is to the LEFT of travel direction
      const kx = Math.cos(((la1 + la2) / 2) * (Math.PI / 180));
      const vx = (lo2 - lo1) * kx;
      const vy = la2 - la1;
      const wx = (lon - footLon) * kx;
      const wy = lat - footLat;
      const cross = vx * wy - vy * wx;
      bestSide = cross > 0 ? 'L' : 'R';
    }
  }

  return {
    mile: Math.round(bestMile * 100) / 100,
    offtrackMi: Math.round(bestDist * 100) / 100,
    side: bestSide,
  };
}

import type { Bundle, Position, Polyline, PositionTableRow } from './types.js';
import { milepostToLatLon, projectToLeg } from './projection.js';

const GPS_STALE_MS = 60_000; // after 60s, switch to dead-reckon

/**
 * PositionService: GPS fix → dead-reckon → predicted-position-table fallback ladder.
 * All methods are synchronous (GPS delivery is caller-managed via onFix).
 */
export class PositionService {
  private legId: string;
  private poly: Polyline;
  private positionTable: PositionTableRow[];

  private lastFixMs: number | null = null;
  private lastFixMile: number | null = null;
  private lastSpeedMph: number = 60; // default train speed
  private lastFixLat: number | null = null;
  private lastFixLon: number | null = null;
  private prevMile: number | null = null;
  private _direction: 1 | -1 = 1;
  private _cached: Position | null = null;

  constructor(bundle: Bundle, poly: Polyline, legId: string) {
    this.legId = legId;
    this.poly = poly;
    this.positionTable = bundle.position_table;
  }

  /** Call with each GPS fix. ts is unix milliseconds. speed is mph (optional). */
  onFix(lat: number, lon: number, ts: number, speed?: number): void {
    const proj = projectToLeg(this.poly, lat, lon);
    const mile = proj.mile;

    if (this.lastFixMile !== null) {
      this._direction = mile >= this.lastFixMile ? 1 : -1;
    }
    this.prevMile = this.lastFixMile;
    this.lastFixMile = mile;
    this.lastFixLat = lat;
    this.lastFixLon = lon;
    this.lastFixMs = ts;
    if (speed !== undefined && speed > 0) this.lastSpeedMph = speed;

    const ll = milepostToLatLon(this.poly, mile);
    this._cached = {
      mile,
      lat: ll.lat,
      lon: ll.lon,
      source: 'gps',
      direction: this._direction,
      leg: this.legId,
    };
  }

  /**
   * Compute position at nowMs.
   * departureMs: optional unix ms of leg departure (for predicted fallback).
   */
  tick(nowMs: number, departureMs?: number): Position {
    // 1. GPS: if we have a recent fix, return it (or extrapolate via dead-reckon)
    if (this.lastFixMs !== null && this.lastFixMile !== null) {
      const elapsedMs = nowMs - this.lastFixMs;
      if (elapsedMs <= GPS_STALE_MS) {
        // Recent enough — return last GPS position
        return this._cached!;
      }
      // Dead-reckon: advance milepost at last known speed
      const elapsedHours = elapsedMs / 3_600_000;
      const deadMile = this.lastFixMile + this._direction * this.lastSpeedMph * elapsedHours;
      const clampedMile = Math.max(
        this.positionTable.length ? this.positionTable[0][1] : 0,
        Math.min(
          this.positionTable.length ? this.positionTable[this.positionTable.length - 1][1] : deadMile,
          deadMile,
        ),
      );
      const ll = milepostToLatLon(this.poly, clampedMile);
      return {
        mile: clampedMile,
        lat: ll.lat,
        lon: ll.lon,
        source: 'deadreckon',
        direction: this._direction,
        leg: this.legId,
      };
    }

    // 2. Predicted: interpolate position_table by elapsed_min from departure
    if (this.positionTable.length >= 2 && departureMs !== undefined) {
      const elapsedMin = (nowMs - departureMs) / 60_000;
      const table = this.positionTable;
      let mile = table[0][1];
      let lat = table[0][2];
      let lon = table[0][3];
      if (elapsedMin >= table[table.length - 1][0]) {
        mile = table[table.length - 1][1];
        lat  = table[table.length - 1][2];
        lon  = table[table.length - 1][3];
      } else {
        for (let i = 0; i < table.length - 1; i++) {
          const [t1, m1, la1, lo1] = table[i];
          const [t2, m2, la2, lo2] = table[i + 1];
          if (t1 <= elapsedMin && elapsedMin <= t2) {
            const f = t2 !== t1 ? (elapsedMin - t1) / (t2 - t1) : 0;
            mile = m1 + f * (m2 - m1);
            lat  = la1 + f * (la2 - la1);
            lon  = lo1 + f * (lo2 - lo1);
            break;
          }
        }
      }
      return { mile, lat, lon, source: 'predicted', direction: 1, leg: this.legId };
    }

    // 3. No data at all — return mile 0 at polyline start
    const start = this.poly.length ? this.poly[0] : [0, 0, 0] as [number, number, number];
    return { mile: start[0], lat: start[1], lon: start[2], source: 'predicted', direction: 1, leg: this.legId };
  }

  current(): Position {
    return this._cached ?? this.tick(Date.now());
  }
}

import type {
  Bundle, Unit, SquibUnit, InterstitialUnit,
  Position, SchedulerSettings, SchedulerResult,
} from './types.js';

const SQUIB_LOOKAHEAD_MI = 1.0;    // fire squib when this far before its mile
const GAP_REENTRY_MI    = 20.0;    // gap in miles that triggers re-entry orientation
const ASSUMED_SPEED_MPH = 60.0;    // fallback speed for gap-time estimation

function isSquib(u: Unit): u is SquibUnit {
  return u.kind === 'squib';
}

function isInterstitial(u: Unit): u is InterstitialUnit {
  return u.kind === 'interstitial';
}

function passesFilter(u: Unit, settings: SchedulerSettings): boolean {
  if (settings.highlightOnly && u.salience < 4) return false;
  if (settings.themes.size > 0 && !settings.themes.has(u.theme)) return false;
  return true;
}

/**
 * Scheduler: given current position and settings, decide what plays now
 * and what goes into the queue.
 *
 * Invariants:
 *  - Squibs fire at most once (tracked in playedIds).
 *  - Interstitials never overrun a squib (fit-or-skip).
 *  - Interstitials are packed greedily by descending salience to hit fillPct.
 */
export class Scheduler {
  private bundle: Bundle;
  private settings: SchedulerSettings;
  private playedIds: Set<string> = new Set();
  private lastMile: number = 0;

  constructor(bundle: Bundle, settings: SchedulerSettings) {
    this.bundle = bundle;
    this.settings = settings;
  }

  select(
    position: Position,
    opts?: { gapMiles?: number },
  ): SchedulerResult {
    const mile = position.mile;
    const gapMiles = opts?.gapMiles ?? (mile - this.lastMile);
    const reEntry = gapMiles > GAP_REENTRY_MI && this.lastMile > 0;
    this.lastMile = mile;

    const units = this.bundle.units;

    // ── 1. Check for due squib ────────────────────────────────────
    const dueSquib = units
      .filter(isSquib)
      .filter(u => !this.playedIds.has(u.id))
      .filter(u => passesFilter(u, this.settings))
      .filter(u => u.mile <= mile + SQUIB_LOOKAHEAD_MI && u.mile >= mile - 2)
      .sort((a, b) => a.mile - b.mile)[0] ?? null;

    if (dueSquib) {
      this.playedIds.add(dueSquib.id);
      return {
        nowPlaying: dueSquib,
        queue: [],
        silenceUntilMile: dueSquib.mile + (dueSquib.dur_s / 3600) * ASSUMED_SPEED_MPH,
      };
    }

    // ── 2. Find next squib to determine gap size ──────────────────
    const nextSquib = units
      .filter(isSquib)
      .filter(u => !this.playedIds.has(u.id))
      .filter(u => passesFilter(u, this.settings))
      .filter(u => u.mile > mile)
      .sort((a, b) => a.mile - b.mile)[0] ?? null;

    const gapEndMile = nextSquib ? nextSquib.mile : (
      this.bundle.position_table.length
        ? this.bundle.position_table[this.bundle.position_table.length - 1][1]
        : mile + 50
    );
    const gapMi = Math.max(0, gapEndMile - mile);
    // Time available in seconds
    const gapSec = (gapMi / ASSUMED_SPEED_MPH) * 3600;
    const fillBudgetSec = gapSec * this.settings.fillPct;

    // ── 3. Pack interstitials into the gap ────────────────────────
    const candidates: InterstitialUnit[] = units
      .filter(isInterstitial)
      .filter(u => !this.playedIds.has(u.id))
      .filter(u => passesFilter(u, this.settings))
      .filter(u => u.from_mi <= mile + SQUIB_LOOKAHEAD_MI && u.to_mi > mile)
      .sort((a, b) => b.salience - a.salience); // descending salience

    const queue: Unit[] = [];
    let usedSec = 0;

    // Re-entry: orient with the highest-salience available interstitial first
    if (reEntry && candidates.length > 0) {
      const orient = candidates[0];
      if (orient.dur_s <= fillBudgetSec) {
        queue.push(orient);
        this.playedIds.add(orient.id);
        usedSec += orient.dur_s;
      }
    }

    for (const u of candidates) {
      if (this.playedIds.has(u.id)) continue;
      if (usedSec + u.dur_s > fillBudgetSec) continue; // fit-or-skip
      queue.push(u);
      this.playedIds.add(u.id);
      usedSec += u.dur_s;
    }

    const nowPlaying = queue.shift() ?? null;
    const silenceUntilMile = nextSquib ? nextSquib.mile : gapEndMile;

    return { nowPlaying, queue, silenceUntilMile };
  }
}

import type { Bundle, EtaResult, Position, PositionTableRow } from './types.js';

/**
 * Eta: compute P10/P50/P90 estimated time of arrival to a mile or station.
 *
 * Strategy: use the bundle's position_table (precomputed elapsed_min → mile mapping)
 * to interpolate how many minutes separate the current mile from the target mile.
 * For a single deterministic table, P10/P50/P90 converge to the same value.
 * Plans 3/4 may replace this with the full statistical engine when online.
 */
export class Eta {
  private table: PositionTableRow[];
  private stationMiles: Map<string, number>;

  constructor(bundle: Bundle) {
    this.table = bundle.position_table;
    // Extract station miles from the guide layer if present
    this.stationMiles = new Map();
    const guide = bundle.layers.guide as Record<string, unknown>;
    if (guide && typeof guide === 'object') {
      const stations = (guide as Record<string, unknown>)['stations'];
      if (Array.isArray(stations)) {
        for (const s of stations) {
          if (typeof s === 'object' && s !== null) {
            const rec = s as Record<string, unknown>;
            if (typeof rec['code'] === 'string' && typeof rec['mile'] === 'number') {
              this.stationMiles.set(rec['code'], rec['mile']);
            }
          }
        }
      }
    }
  }

  /** Interpolate elapsed_min at a given mile from the position table. */
  private elapsedMinAtMile(targetMile: number): number | null {
    const t = this.table;
    if (t.length === 0) return null;
    if (targetMile <= t[0][1]) return t[0][0];
    if (targetMile >= t[t.length - 1][1]) return t[t.length - 1][0];
    for (let i = 0; i < t.length - 1; i++) {
      const [e1, m1] = t[i];
      const [e2, m2] = t[i + 1];
      if (m1 <= targetMile && targetMile <= m2) {
        const f = m2 !== m1 ? (targetMile - m1) / (m2 - m1) : 0;
        return e1 + f * (e2 - e1);
      }
    }
    return null;
  }

  toMile(targetMile: number, currentPosition: Position): EtaResult {
    const currentElapsed = this.elapsedMinAtMile(currentPosition.mile);
    const targetElapsed  = this.elapsedMinAtMile(targetMile);

    if (currentElapsed === null || targetElapsed === null) {
      return { p10: NaN, p50: NaN, p90: NaN };
    }
    // Simple deterministic estimate; spread a small uncertainty window (±5%)
    const eta = Math.max(0, targetElapsed - currentElapsed);
    const spread = eta * 0.05;
    return {
      p10: Math.max(0, eta - spread),
      p50: eta,
      p90: eta + spread,
    };
  }

  toStation(code: string, currentPosition: Position): EtaResult {
    const mile = this.stationMiles.get(code);
    if (mile === undefined) {
      return { p10: -1, p50: -1, p90: -1 };
    }
    return this.toMile(mile, currentPosition);
  }
}

import type { Bundle, Unit, DiveGrounding } from './types.js';

/**
 * Assemble grounding context for an LLM dive from the bundle's data layers.
 * Returns the unit's text + relevant slices of connections, lore, science, theme.
 * Used by the online dive feature in Plan 3/4.
 */
export function diveGrounding(
  bundle: Bundle,
  unitId: string,
  focus?: string,
): DiveGrounding {
  const unit: Unit | undefined = bundle.units.find(u => u.id === unitId);
  if (!unit) {
    throw new Error(`diveGrounding: unit "${unitId}" not found in bundle leg ${bundle.leg}`);
  }

  const layers = bundle.layers;

  // ── Theme slice ──────────────────────────────────────────────────
  const themes = layers.themes as Record<string, unknown>;
  const theme = unit.theme in themes ? themes[unit.theme] : null;

  // ── Lore slice — find entries near unit's mile ───────────────────
  const loreData = layers.lore as Record<string, unknown>;
  const loreNodes = Array.isArray((loreData as Record<string, unknown>)['lore'])
    ? (loreData as Record<string, unknown>)['lore'] as Array<Record<string, unknown>>
    : [];
  const unitMile = unit.kind === 'squib' ? unit.mile : (unit.from_mi + unit.to_mi) / 2;
  const nearbyLore = loreNodes
    .filter(n => {
      const m = n['peak_mi'];
      return typeof m === 'number' && Math.abs(m - unitMile) < 20;
    })
    .slice(0, 3);

  // ── Connections slice — nodes mentioning the unit's place ────────
  const connData = layers.connections as Record<string, unknown>;
  const connNodes = typeof connData['nodes'] === 'object' && connData['nodes'] !== null
    ? connData['nodes'] as Record<string, unknown>
    : {};
  const relevantConn = Object.values(connNodes)
    .filter((n): n is Record<string, unknown> => typeof n === 'object' && n !== null)
    .filter(n => {
      const title = String(n['title'] ?? '');
      return title.includes(unit.place.split(',')[0]);
    })
    .slice(0, 3);

  // ── Science slice ────────────────────────────────────────────────
  const scienceData = layers.science as Record<string, unknown>;
  const sciIntervals = Array.isArray(scienceData['intervals'])
    ? (scienceData['intervals'] as Array<Record<string, unknown>>)
        .filter(iv => {
          const lo = Number(iv['mile_lo'] ?? 0);
          const hi = Number(iv['mile_hi'] ?? 0);
          return lo <= unitMile && unitMile <= hi;
        })
        .slice(0, 2)
    : [];

  // ── Sources list ─────────────────────────────────────────────────
  const sources: string[] = [`Unit text: ${unit.text}`];
  if (theme && typeof (theme as Record<string, unknown>)['thesis'] === 'string') {
    sources.push(`Theme: ${(theme as Record<string, unknown>)['thesis']}`);
  }
  for (const ln of nearbyLore) {
    if (typeof ln['title'] === 'string') sources.push(`Lore: ${ln['title']}`);
  }
  for (const cn of relevantConn) {
    if (typeof cn['title'] === 'string') sources.push(`Connection: ${cn['title']}`);
  }
  if (focus) {
    sources.push(`User focus: ${focus}`);
  }

  return {
    unitText: unit.text,
    connections: relevantConn,
    lore: nearbyLore,
    science: sciIntervals,
    theme,
    sources,
  };
}

  StorageAdapter, Dive, Position,
} from './types.js';

let _idCounter = 0;
function generateId(): string {
  return `fav-${Date.now()}-${++_idCounter}`;
}

function snapshotUnit(unit: Unit): UnitSnapshot {
  return {
    kind: unit.kind,
    mile: unit.kind === 'squib' ? unit.mile : undefined,
    from_mi: unit.kind === 'interstitial' ? unit.from_mi : undefined,
    to_mi: unit.kind === 'interstitial' ? unit.to_mi : undefined,
    place: unit.place,
    theme: unit.theme,
    text: unit.text,
  };
}

/**
 * In-memory StorageAdapter for tests and offline-first use before native storage
 * is wired in Plan 3.
 */
export class InMemoryStorageAdapter implements StorageAdapter {
  private store: Map<string, Favorite> = new Map();

  async save(favorite: Favorite): Promise<void> {
    this.store.set(favorite.id, { ...favorite });
  }

  async loadAll(): Promise<Favorite[]> {
    return Array.from(this.store.values());
  }

  async loadById(id: string): Promise<Favorite | null> {
    return this.store.get(id) ?? null;
  }

  async update(id: string, patch: Partial<Favorite>): Promise<void> {
    const existing = this.store.get(id);
    if (!existing) throw new Error(`InMemoryStorageAdapter: favorite "${id}" not found`);
    this.store.set(id, { ...existing, ...patch });
  }
}

/**
 * Favorites: capture, retrieve, and annotate unit favorites.
 * Works against any StorageAdapter (InMemoryStorageAdapter for tests,
 * CapacitorStorageAdapter in Plan 3).
 */
export class Favorites {
  private adapter: StorageAdapter;

  constructor(adapter: StorageAdapter) {
    this.adapter = adapter;
  }

  async add(
    unit: Unit,
    leg: string,
    position: Position,
    kind: FavoriteKind,
    note?: string,
  ): Promise<Favorite> {
    const fav: Favorite = {
      id: generateId(),
      timestamp: Date.now(),
      leg,
      unitSnapshot: snapshotUnit(unit),
      lat: position.lat,
      lon: position.lon,
      kind,
      note,
    };
    await this.adapter.save(fav);
    return fav;
  }

  async list(): Promise<Favorite[]> {
    return this.adapter.loadAll();
  }

  async get(id: string): Promise<Favorite | null> {
    return this.adapter.loadById(id);
  }

  async attachDive(id: string, dive: Dive): Promise<void> {
    const existing = await this.adapter.loadById(id);
    if (!existing) throw new Error(`Favorites.attachDive: favorite "${id}" not found`);
    await this.adapter.update(id, { dive });
  }
}
