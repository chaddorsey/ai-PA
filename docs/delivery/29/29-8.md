## Requirements

1. **Rate Limiting**
   - Maximum 4 insights per minute (configurable)
   - Minimum 8-second gap between insights (configurable)
   - Immediate bypass for high-priority events (scores, turnovers)

2. **Priority Queue**
   - Priority 1-10 scale (higher = more important)
   - Higher priority insights delivered first
   - TTL expiration for stale insights

3. **Game State Adaptation**
   - Reduce frequency during intense action
   - Increase during stoppages and commercial breaks
   - Queue insights during active plays

4. **User Engagement Signals**
   - Track time since last user query
   - Infer engagement from interaction patterns
   - Adjust frequency accordingly

5. **Game Stakes Consideration**
   - Track score differential
   - Adjust for rivalry/importance signals
   - Close games = fewer interruptions

# delivery_manager.py

from dataclasses import dataclass
from datetime import datetime
from typing import Optional, Callable, List
import asyncio
import time

@dataclass
class Insight:
    id: str
    type: str
    priority: int  # 1-10
    timing: str  # immediate, post_play, stoppage
    headline: str
    body: str
    ttl: int  # seconds
    generated_at: datetime
    delivered: bool = False
    delivered_at: Optional[datetime] = None

    def is_expired(self) -> bool:
        age = (datetime.now() - self.generated_at).total_seconds()
        return age > self.ttl

    def __lt__(self, other):
        # For priority queue - higher priority = lower number for min-heap
        return self.priority > other.priority


class DeliveryConfig:
    def __init__(self):
        self.max_insights_per_minute: int = 4
        self.min_gap_seconds: float = 8.0
        self.immediate_priority_threshold: int = 9  # 9-10 = immediate
        self.high_action_multiplier: float = 2.0  # Increase gap during action
        self.low_action_multiplier: float = 0.5  # Decrease gap during stoppages


class GameStateTracker:
    """Track game state for delivery timing decisions."""
    
    def __init__(self):
        self.clock_running: bool = False
        self.is_commercial: bool = False
        self.is_timeout: bool = False
        self.is_halftime: bool = False
        self.score_differential: int = 0
        self.quarter: int = 1
        self.time_since_last_play: float = 0
        self.last_play_timestamp: datetime = datetime.now()
    
    def update(self, game_state: dict):
        self.clock_running = game_state.get('clock_running', False)
        self.is_timeout = game_state.get('is_timeout', False)
        self.is_halftime = game_state.get('status') == 'halftime'
        self.quarter = game_state.get('quarter', 1)
        
        home_score = game_state.get('home_score', 0)
        away_score = game_state.get('away_score', 0)
        self.score_differential = abs(home_score - away_score)
    
    def get_action_level(self) -> str:
        """Determine current game action level."""
        if self.is_halftime:
            return 'minimal'
        if self.is_timeout or self.is_commercial:
            return 'low'
        if not self.clock_running:
            return 'moderate'
        return 'high'
    
    def get_stakes_level(self) -> str:
        """Determine game stakes based on score and time."""
        if self.quarter >= 4 and self.score_differential <= 7:
            return 'high'  # Close game in 4th quarter
        if self.score_differential >= 21:
            return 'low'  # Blowout
        return 'moderate'


class UserEngagementTracker:
    """Track user engagement signals."""
    
    def __init__(self):
        self.last_query_time: Optional[datetime] = None
        self.queries_last_10_min: int = 0
        self.insights_acknowledged: int = 0
        self.session_start: datetime = datetime.now()
    
    def record_query(self):
        self.last_query_time = datetime.now()
        self.queries_last_10_min += 1
    
    def get_engagement_level(self) -> str:
        """Infer engagement from interaction patterns."""
        if self.last_query_time is None:
            return 'passive'
        
        since_last = (datetime.now() - self.last_query_time).total_seconds()
        
        if since_last < 60:
            return 'active'  # Recent interaction
        if self.queries_last_10_min >= 3:
            return 'engaged'  # Multiple recent queries
        return 'passive'


class DeliveryManager:
    """Manage insight delivery with adaptive timing."""
    
    def __init__(self, config: DeliveryConfig):
        self.config = config
        self.queue: asyncio.PriorityQueue = asyncio.PriorityQueue()
        self.recent_deliveries: List[datetime] = []
        self.last_delivery_time: float = 0
        self.game_tracker = GameStateTracker()
        self.user_tracker = UserEngagementTracker()
        self.delivered_ids: set = set()
    
    async def queue_insight(self, insight: Insight):
        """Add insight to priority queue."""
        if insight.id in self.delivered_ids:
            return  # Duplicate
        
        # Immediate delivery for critical insights
        if insight.priority >= self.config.immediate_priority_threshold:
            insight.timing = 'immediate'
        
        await self.queue.put(insight)
    
    async def delivery_loop(self, send_callback: Callable):
        """Main delivery loop."""
        while True:
            try:
                # Check if we can deliver
                if await self._can_deliver_now():
                    insight = await self._get_next_deliverable()
                    if insight:
                        await send_callback(insight)
                        self._record_delivery(insight)
                
                await asyncio.sleep(0.5)
                
            except Exception as e:
                print(f"Delivery error: {e}")
                await asyncio.sleep(1.0)
    
    async def _can_deliver_now(self) -> bool:
        """Check if delivery conditions are met."""
        now = time.time()
        
        # Calculate adaptive gap based on game state
        base_gap = self.config.min_gap_seconds
        action_level = self.game_tracker.get_action_level()
        
        if action_level == 'high':
            # During active play, increase gap significantly
            gap = base_gap * self.config.high_action_multiplier
        elif action_level == 'low' or action_level == 'minimal':
            # During stoppages, can be more frequent
            gap = base_gap * self.config.low_action_multiplier
        else:
            gap = base_gap
        
        # Adjust for stakes
        stakes = self.game_tracker.get_stakes_level()
        if stakes == 'high':
            # Close game = less interruption
            gap *= 1.5
        
        # Check minimum gap
        if now - self.last_delivery_time < gap:
            return False
        
        # Check rate limit
        recent = [t for t in self.recent_deliveries 
                  if (now - t.timestamp()) < 60]
        if len(recent) >= self.config.max_insights_per_minute:
            return False
        
        return True
    
    async def _get_next_deliverable(self) -> Optional[Insight]:
        """Get next insight that's ready to deliver."""
        try:
            insight = await asyncio.wait_for(
                self.queue.get(),
                timeout=0.1
            )
            
            # Check if expired
            if insight.is_expired():
                return None
            
            # Check timing
            if insight.timing == 'immediate':
                return insight
            
            action_level = self.game_tracker.get_action_level()
            
            if insight.timing == 'post_play' and action_level == 'high':
                # Re-queue for later
                await self.queue.put(insight)
                return None
            
            return insight
            
        except asyncio.TimeoutError:
            return None
    
    def _record_delivery(self, insight: Insight):
        """Record that an insight was delivered."""
        now = datetime.now()
        self.last_delivery_time = time.time()
        self.recent_deliveries.append(now)
        self.delivered_ids.add(insight.id)
        insight.delivered = True
        insight.delivered_at = now
        
        # Cleanup old records
        cutoff = now.timestamp() - 120
        self.recent_deliveries = [
            t for t in self.recent_deliveries 
            if t.timestamp() > cutoff
        ]
    
    def update_game_state(self, state: dict):
        """Update game state for timing decisions."""
        self.game_tracker.update(state)
    
    def record_user_query(self):
        """Record user interaction."""
        self.user_tracker.record_query()
