import asyncio
import aiohttp
import pytest
from datetime import datetime

BASE_URL = "http://localhost:5130"
INSIGHT_ENGINE_URL = "http://localhost:5131"
GAME_STATE_URL = "http://localhost:5132"


class TestAutoMaddenE2E:
    """End-to-end tests for Auto-Madden system."""
    
    @pytest.mark.asyncio
    async def test_health_checks(self):
        """Verify all services are healthy."""
        async with aiohttp.ClientSession() as session:
            # UI health
            async with session.get(f"{BASE_URL}/health") as resp:
                assert resp.status == 200
            
            # Insight engine health
            async with session.get(f"{INSIGHT_ENGINE_URL}/health") as resp:
                assert resp.status == 200
            
            # Game state health
            async with session.get(f"{GAME_STATE_URL}/health") as resp:
                assert resp.status == 200
    
    @pytest.mark.asyncio
    async def test_start_session(self):
        """Test starting a game session."""
        async with aiohttp.ClientSession() as session:
            # List available games
            async with session.get(f"{GAME_STATE_URL}/games") as resp:
                assert resp.status == 200
                games = await resp.json()
                assert 'games' in games
            
            # Start session (with first available game or specific team)
            async with session.post(
                f"{GAME_STATE_URL}/start",
                json={"team": "patriots"}
            ) as resp:
                result = await resp.json()
                # May not find game if no Patriots playing
                assert 'status' in result
    
    @pytest.mark.asyncio
    async def test_websocket_connection(self):
        """Test WebSocket connection and message receipt."""
        async with aiohttp.ClientSession() as session:
            async with session.ws_connect(f"{BASE_URL}/ws") as ws:
                # Should receive initial connection message
                msg = await asyncio.wait_for(ws.receive(), timeout=5)
                assert msg.type == aiohttp.WSMsgType.TEXT
                
                # Send start session command
                await ws.send_json({
                    "type": "start",
                    "query": "patriots"
                })
                
                # Should receive response
                msg = await asyncio.wait_for(ws.receive(), timeout=10)
                assert msg.type == aiohttp.WSMsgType.TEXT
    
    @pytest.mark.asyncio
    async def test_user_query(self):
        """Test asking a question."""
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{INSIGHT_ENGINE_URL}/query",
                json={"question": "What's an RPO?"}
            ) as resp:
                assert resp.status == 200
                result = await resp.json()
                assert 'answer' in result
    
    @pytest.mark.asyncio
    async def test_insight_delivery(self):
        """Test that insights are delivered within rate limits."""
        insights_received = []
        start_time = datetime.now()
        
        async with aiohttp.ClientSession() as session:
            async with session.ws_connect(f"{BASE_URL}/ws") as ws:
                # Start session
                await ws.send_json({"type": "start", "query": "test"})
                
                # Collect insights for 60 seconds
                try:
                    while (datetime.now() - start_time).seconds < 60:
                        msg = await asyncio.wait_for(ws.receive(), timeout=15)
                        if msg.type == aiohttp.WSMsgType.TEXT:
                            data = msg.json()
                            if data.get('type') == 'insight':
                                insights_received.append({
                                    'time': datetime.now(),
                                    'insight': data
                                })
                except asyncio.TimeoutError:
                    pass
        
        # Verify rate limiting
        if len(insights_received) > 1:
            for i in range(1, len(insights_received)):
                gap = (insights_received[i]['time'] - 
                       insights_received[i-1]['time']).total_seconds()
                assert gap >= 7.0, f"Gap too short: {gap}s"
        
        assert len(insights_received) <= 4, "Too many insights per minute"

### README Documentation

# Auto-Madden: Real-Time Game Companion

Auto-Madden is an AI-powered companion for watching NFL games. It provides
real-time insights, answers questions, and helps you understand the game
better—like having John Madden on your couch.

## Quick Start

1. **Start the services:**
   ```bash
   docker-compose up auto-madden-companion-ui auto-madden-insight-engine auto-madden-game-state
   ```

2. **Open the companion UI:**
   Navigate to http://localhost:5130

3. **Start a game session:**
   Type "Watch the Patriots game" (or any team playing)

4. **Watch and learn:**
   - Insights push automatically during the game
   - Ask questions anytime: "What's an RPO?"
   - Say "end session" when done

## Architecture

│   (Port 5130)   │    │   (Port 5131)   │    │   (Port 5132)   │
└─────────────────┘    └────────┬────────┘    └────────┬────────┘

## Configuration

Environment variables:
- `LLM_PROVIDER`: anthropic, openai (default: anthropic)
- `LLM_MODEL`: Model to use for insights
- `MAX_INSIGHTS_PER_MINUTE`: Rate limit (default: 4)
- `MIN_INSIGHT_GAP_SECONDS`: Minimum gap (default: 8)

## Troubleshooting

**No insights appearing:**
- Check game-state-service health: `curl localhost:5132/health`
- Verify a game is active: `curl localhost:5132/state`

**Connection drops:**
- UI auto-reconnects after 3 seconds
- Check insight-engine logs for errors

**Questions not answered:**
- Verify Letta is running: `curl localhost:8283/v1/health/`
- Check agent has auto-madden tools attached
