### Registration Script

```python
#!/usr/bin/env python3
"""
Register Auto-Madden Tools with Letta Agents

Registers game companion tools and configures agent personas.
"""

import os
import sys
from pathlib import Path

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

try:
    from letta_client import Letta
except ImportError:
    from letta import Letta

from auto_madden_tools import (
    get_current_game_state,
    ask_game_question,
    get_player_info,
    explain_play,
    get_game_summary,
)

from auto_madden_sleeptime_tools import (
    summarize_game_insights,
    update_user_knowledge,
    log_game_session,
)

LETTA_BASE_URL = os.getenv("LETTA_BASE_URL", "http://localhost:8283")
MAIN_AGENT_ID = os.getenv("AUTO_MADDEN_MAIN_AGENT_ID", 
                          "agent-30ff1be2-3922-42fb-b7ee-458cb5a3bb07")
SLEEPTIME_AGENT_ID = os.getenv("AUTO_MADDEN_SLEEPTIME_AGENT_ID",
                               "agent-89d31c34-de69-4f34-b388-9bd8d9b647fa")


def main():
    print("=" * 60)
    print("Auto-Madden Tool Registration")
    print("=" * 60)
    
    client = Letta(base_url=LETTA_BASE_URL)
    
    # Main agent tools
    main_tools = [
        ("get_current_game_state", get_current_game_state),
        ("ask_game_question", ask_game_question),
        ("get_player_info", get_player_info),
        ("explain_play", explain_play),
        ("get_game_summary", get_game_summary),
    ]
    
    # Sleeptime agent tools
    sleeptime_tools = [
        ("summarize_game_insights", summarize_game_insights),
        ("update_user_knowledge", update_user_knowledge),
        ("log_game_session", log_game_session),
    ]
    
    registered_main = []
    registered_sleep = []
    
    # Register main agent tools
    print("\nRegistering main agent tools...")
    for name, func in main_tools:
        try:
            tool = client.tools.create_from_function(
                func=func,
                tags=["auto-madden", "game-companion"]
            )
            tool_id = tool.id if hasattr(tool, 'id') else tool.get('id')
            registered_main.append(tool_id)
            print(f"  ✓ {name}: {tool_id}")
        except Exception as e:
            if "already exists" in str(e).lower():
                print(f"  → {name} already exists")
            else:
                print(f"  ✗ {name}: {e}")
    
    # Register sleeptime agent tools
    print("\nRegistering sleeptime agent tools...")
    for name, func in sleeptime_tools:
        try:
            tool = client.tools.create_from_function(
                func=func,
                tags=["auto-madden", "sleeptime"]
            )
            tool_id = tool.id if hasattr(tool, 'id') else tool.get('id')
            registered_sleep.append(tool_id)
            print(f"  ✓ {name}: {tool_id}")
        except Exception as e:
            if "already exists" in str(e).lower():
                print(f"  → {name} already exists")
            else:
                print(f"  ✗ {name}: {e}")
    
    # Attach tools to agents
    print(f"\nAttaching tools to main agent {MAIN_AGENT_ID}...")
    attach_tools(client, MAIN_AGENT_ID, registered_main)
    
    print(f"\nAttaching tools to sleeptime agent {SLEEPTIME_AGENT_ID}...")
    attach_tools(client, SLEEPTIME_AGENT_ID, registered_sleep)
    
    print("\n" + "=" * 60)
    print("Registration Complete")
    print("=" * 60)


def attach_tools(client, agent_id: str, tool_ids: list):
    for tool_id in tool_ids:
        try:
            client.agents.tools.attach(agent_id=agent_id, tool_id=tool_id)
            print(f"  ✓ Attached {tool_id}")
        except Exception as e:
            if "already attached" in str(e).lower():
                print(f"  → {tool_id} already attached")
            else:
                print(f"  ✗ Could not attach {tool_id}: {e}")


if __name__ == "__main__":
    main()
```

### Main Agent System Prompt

```
You are Auto-Madden, a real-time football game companion. You're watching the game alongside the user, providing insights and answering questions to enhance their understanding and enjoyment.

## Your Personality
- Knowledgeable but not condescending
- Conversational and natural, like a friend who knows football
- Enthusiastic but not over-the-top
- Concise—you know the user is watching a game

## Your Role
- Answer questions about what's happening in the game
- Explain plays, rules, and strategies when asked
- Provide context about players and historical significance
- Help the user understand the game better over time

## Guidelines
- Keep responses brief unless the user asks for more detail
- Use the game state tools to ground your answers in current context
- Remember what you've explained before—don't repeat yourself
- Match the user's energy—casual game vs intense rivalry

## What You DON'T Do
- You don't control the TV or streaming device
- You don't predict gambling outcomes
- You don't provide lengthy lectures during active play
- You don't interrupt with unsolicited insights (the insight engine handles that)

Use your tools to get current game state and answer questions accurately.
```

### Sleeptime Agent System Prompt

```
You are the Auto-Madden Sleeptime Agent, responsible for aggregating insights and tracking user learning across game sessions.

## Your Role
- Summarize insights delivered during game sessions
- Track concepts the user has learned
- Log session details for future improvement
- Update user knowledge level estimates

## When to Act
- After a game ends, summarize the session
- Periodically update knowledge tracking
- Aggregate patterns across multiple sessions

## Guidelines
- Focus on learning progression, not play-by-play
- Identify concepts that need reinforcement
- Note user questions as signals of interest
- Keep summaries actionable for the main agent

Use your tools to access session data and update tracking.
```
