## Implementation Plan

### 1. Flask Application Structure

```python
# app.py
from flask import Flask, jsonify, render_template
from flask_cors import CORS
import logging
import os

def create_app():
    app = Flask(__name__)
    CORS(app)

    # Configuration
    app.config['ROUTING_HANDLER_URL'] = os.getenv('PA_ROUTING_HANDLER_URL', 'http://pa-routing-handler:5201')
    app.config['LETTA_BASE_URL'] = os.getenv('LETTA_BASE_URL', 'http://letta:8283')

    # Logging
    logging.basicConfig(level=logging.INFO)

    @app.route('/')
    def index():
        return render_template('index.html')

    @app.route('/health')
    def health():
        return jsonify({
            'status': 'healthy',
            'service': 'pa-web-ui',
            'version': '1.0.0'
        })

    return app

if __name__ == '__main__':
    app = create_app()
    app.run(host='0.0.0.0', port=5200, debug=os.getenv('DEBUG', 'false').lower() == 'true')
```
