## Implementation Plan

### 1. Schema Design

```sql
-- Create schema
CREATE SCHEMA IF NOT EXISTS pa_web;

-- Conversations table
CREATE TABLE pa_web.conversations (
    id SERIAL PRIMARY KEY,
    session_id UUID NOT NULL,
    user_id VARCHAR(255),
    agent_id VARCHAR(255) NOT NULL,
    agent_name VARCHAR(255),
    role VARCHAR(20) NOT NULL,
    message TEXT NOT NULL,
    metadata JSONB,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_conv_session_id ON pa_web.conversations(session_id);
CREATE INDEX idx_conv_created_at ON pa_web.conversations(created_at DESC);

-- Routing decisions table
CREATE TABLE pa_web.routing_decisions (
    id SERIAL PRIMARY KEY,
    session_id UUID NOT NULL,
    message_preview VARCHAR(255),
    selected_agent_id VARCHAR(255) NOT NULL,
    routing_method VARCHAR(50),
    routing_confidence FLOAT,
    processing_time_ms INT,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_routing_session ON pa_web.routing_decisions(session_id);

-- Sessions table
CREATE TABLE pa_web.sessions (
    session_id UUID PRIMARY KEY,
    user_id VARCHAR(255),
    current_agent_id VARCHAR(255),
    preferences JSONB,
    last_activity TIMESTAMP DEFAULT NOW(),
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_sessions_last_activity ON pa_web.sessions(last_activity);
```
