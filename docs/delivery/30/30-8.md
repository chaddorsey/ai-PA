```python
# app.py additions
from flask import Response, request, stream_with_context
import json
import uuid
import httpx

sessions = {}  # In-memory session store

@app.route('/stream', methods=['POST'])
def stream():
    data = request.json
    message = data.get('message', '')
    agent_id = data.get('agent_id')
    session_id = data.get('session_id') or str(uuid.uuid4())

    def generate():
        try:
            # 1. Route to get agent
            routing_response = route_message(session_id, message, agent_id)
            selected_agent = routing_response['agent_id']

            yield f"data: {json.dumps({'type': 'routing', 'agent_name': routing_response['agent_name']})}\n\n"

            # 2. Stream from Letta
            full_response = ""
            for event in stream_letta(selected_agent, message):
                if event['type'] == 'text':
                    full_response += event['content']
                yield f"data: {json.dumps(event)}\n\n"

            # 3. Save to history
            save_conversation(session_id, selected_agent, message, full_response)

            yield f"data: {json.dumps({'type': 'done', 'session_id': session_id})}\n\n"

        except Exception as e:
            yield f"data: {json.dumps({'type': 'error', 'message': str(e)})}\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype='text/event-stream',
        headers={
            'Cache-Control': 'no-cache',
            'Connection': 'keep-alive',
            'X-Accel-Buffering': 'no'
        }
    )

def route_message(session_id, message, agent_id):
    """Call routing handler"""
    response = httpx.post(
        f"{app.config['ROUTING_HANDLER_URL']}/v1/route",
        json={
            "session_id": session_id,
            "message": message,
            "agent_id": agent_id
        }
    )
    return response.json()

def stream_letta(agent_id, message):
    """Stream from Letta and yield events"""
    # Use LettaClient from task 30-7
    client = LettaClient(app.config['LETTA_BASE_URL'])
    for event in client.stream_message(agent_id, message):
        yield event
```

### 2. Multi-Agent Orchestrator (Patterns 2, 3, 4)

```python
# orchestrator.py
import asyncio
from datetime import datetime
from pa_routing.models.session import SessionContext
from pa_routing.services.session_store import session_store
from pa_routing.services.summary_parser import extract_summary, clean_response_for_user
from pa_routing.services.letta_client import LettaClient

class MultiAgentOrchestrator:
    """
    Orchestrates multi-agent conversations with context sharing.
    No LLM at handler level - all logic is deterministic Python.
    """

    def __init__(self, letta_client: LettaClient, main_agent_id: str):
        self.client = letta_client
        self.main_agent_id = main_agent_id

    async def handle_message(
        self,
        user_message: str,
        user_id: str,
        explicit_agent_id: str = None
    ) -> tuple[str, str, list]:
        """
        Handle user message with full orchestration.

        Returns: (clean_response, agent_name, events_for_streaming)
        """
        session = session_store.get_or_create(user_id)

        # 1. Route to appropriate agent
        agent_id, agent_name, confidence = self._route_message(
            user_message, explicit_agent_id
        )
        is_main_agent = (agent_id == self.main_agent_id)

        # 2. Build message with context injection (Pattern 2)
        context_prefix = session.format_for_injection()

        # Pattern 4: Briefing injection (main agent only)
        if is_main_agent:
            briefing = await self._get_session_briefing()
            full_message = self._build_message(briefing, context_prefix, user_message)
        else:
            full_message = self._build_message("", context_prefix, user_message)

        # 3. Stream from agent and collect response + tool calls
        response_text = ""
        tool_calls = []
        events = []

        async for event in self.client.stream_message(agent_id, full_message):
            events.append(event)
            if event["type"] == "text":
                response_text += event["content"]
            elif event["type"] == "tool_call":
                tool_calls.append(event["tool_name"])

        # 4. Parse SUMMARY and update session context (Pattern 2)
        summary = extract_summary(response_text, agent_name, tool_calls)
        session.append(agent=agent_name, action=summary)

        # 5. Write to archival (Pattern 3 - fire-and-forget for sub-agents)
        if not is_main_agent:
            asyncio.create_task(self._write_to_archival(
                user_id=user_id,
                agent_name=agent_name,
                message=user_message,
                summary=summary
            ))

        # 6. Return clean response
        clean_response = clean_response_for_user(response_text)
        return clean_response, agent_name, events

    def _route_message(self, message: str, explicit_agent_id: str = None):
        """Deterministic routing - delegates to AgentSelector from 30-6"""
        from pa_routing.services.agent_selector import AgentSelector
        selector = AgentSelector(self.main_agent_id)
        return selector.select(message, explicit_agent_id)

    def _build_message(self, briefing: str, context: str, user_message: str) -> str:
        """Combine briefing, context, and user message."""
        parts = [p for p in [briefing, context, user_message] if p]
        return "\n\n".join(parts)

    async def _get_session_briefing(self) -> str:
        """Pattern 4: Query recent passages for briefing"""
        today = datetime.utcnow().strftime("%Y-%m-%d")
        passages = await self.client.list_passages(
            agent_id=self.main_agent_id,
            tags=[f"session:{today}"],
            limit=5
        )
        return self.client.format_briefing(passages)

    async def _write_to_archival(
        self,
        user_id: str,
        agent_name: str,
        message: str,
        summary: str
    ):
        """Pattern 3: Async archival persistence (fire-and-forget)"""
        today = datetime.utcnow().strftime("%Y-%m-%d")

        # Simple topic extraction from first word
        topic = message.split()[0].lower() if message else "general"

        passage_text = f"User asked {agent_name}: {message[:80]}... Action: {summary}"

        try:
            await self.client.create_passage(
                agent_id=self.main_agent_id,
                text=passage_text,
                tags=[
                    f"session:{today}",
                    f"agent:{agent_name}",
                    f"topic:{topic}",
                    f"user:{user_id}"
                ]
            )
        except Exception as e:
            # Fire-and-forget: log but don't fail
            import structlog
            structlog.get_logger().warning("archival_write_failed", error=str(e))
```

```python
# app.py - updated stream endpoint
from orchestrator import MultiAgentOrchestrator

# Initialize once at startup
orchestrator = MultiAgentOrchestrator(
    letta_client=LettaClient(app.config['LETTA_BASE_URL']),
    main_agent_id=app.config['MAIN_AGENT_ID']
)

@app.route('/stream', methods=['POST'])
async def stream():
    data = request.json
    message = data.get('message', '')
    agent_id = data.get('agent_id')
    user_id = data.get('user_id', 'anonymous')
    session_id = data.get('session_id') or str(uuid.uuid4())

    async def generate():
        try:
            # Use orchestrator for full context-aware handling
            clean_response, agent_name, events = await orchestrator.handle_message(
                user_message=message,
                user_id=user_id,
                explicit_agent_id=agent_id
            )

            # Emit routing event
            yield f"data: {json.dumps({'type': 'routing', 'agent_name': agent_name})}\n\n"

            # Stream collected events to frontend
            for event in events:
                yield f"data: {json.dumps(event)}\n\n"

            # Save conversation to database
            save_conversation(session_id, agent_name, message, clean_response)

            yield f"data: {json.dumps({'type': 'done', 'session_id': session_id})}\n\n"

        except Exception as e:
            yield f"data: {json.dumps({'type': 'error', 'message': str(e)})}\n\n"

    return Response(
        stream_with_context(generate()),
        mimetype='text/event-stream',
        headers={
            'Cache-Control': 'no-cache',
            'Connection': 'keep-alive',
            'X-Accel-Buffering': 'no'
        }
    )
```

```javascript
    const reader = response.body.getReader();
```
