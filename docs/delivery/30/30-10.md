### 1. Upload Endpoint

```python
# app.py additions
from werkzeug.utils import secure_filename
import base64

ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg', 'gif', 'pdf', 'txt', 'md'}
MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/api/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'error': 'No file provided'}), 400

    file = request.files['file']
    if file.filename == '':
        return jsonify({'error': 'No file selected'}), 400

    if not allowed_file(file.filename):
        return jsonify({'error': 'File type not allowed'}), 400

    # Read and encode
    content = file.read()
    if len(content) > MAX_FILE_SIZE:
        return jsonify({'error': f'File too large (max {MAX_FILE_SIZE // 1024 // 1024}MB)'}), 400

    encoded = base64.b64encode(content).decode('utf-8')
    filename = secure_filename(file.filename)

    return jsonify({
        'filename': filename,
        'mime_type': file.content_type,
        'size': len(content),
        'data': encoded
    })
```

### 3. JavaScript Upload Handler

```javascript
// upload.js
class FileUploader {
    constructor() {
        this.files = [];
        this.setupDragDrop();
        this.setupFileInput();
    }

    setupDragDrop() {
        const dropZone = document.getElementById('file-drop-zone');

        dropZone.addEventListener('dragover', (e) => {
            e.preventDefault();
            dropZone.classList.add('drag-over');
        });

        dropZone.addEventListener('dragleave', () => {
            dropZone.classList.remove('drag-over');
        });

        dropZone.addEventListener('drop', (e) => {
            e.preventDefault();
            dropZone.classList.remove('drag-over');
            this.handleFiles(e.dataTransfer.files);
        });
    }

    async handleFiles(fileList) {
        for (const file of fileList) {
            if (file.size > 10 * 1024 * 1024) {
                this.showError('File too large (max 10MB)');
                continue;
            }

            const formData = new FormData();
            formData.append('file', file);

            const response = await fetch('/api/upload', {
                method: 'POST',
                body: formData
            });

            if (response.ok) {
                const data = await response.json();
                this.files.push(data);
                this.showPreview(data);
            }
        }
    }

    showPreview(fileData) {
        const preview = document.getElementById('file-preview');
        const item = document.createElement('div');
        item.className = 'file-preview-item';

        if (fileData.mime_type.startsWith('image/')) {
            item.innerHTML = `
                <img src="data:${fileData.mime_type};base64,${fileData.data}" />
                <span>${fileData.filename}</span>
                <button onclick="fileUploader.removeFile('${fileData.filename}')">×</button>
            `;
        } else {
            item.innerHTML = `
                <span class="file-icon">📄</span>
                <span>${fileData.filename}</span>
                <button onclick="fileUploader.removeFile('${fileData.filename}')">×</button>
            `;
        }

        preview.appendChild(item);
    }

    getFilesForMessage() {
        const files = this.files;
        this.files = [];
        document.getElementById('file-preview').innerHTML = '';
        return files;
    }
}
```
