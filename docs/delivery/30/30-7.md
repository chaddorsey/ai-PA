### 1. LettaClient Class

```python
# src/pa_routing/services/letta_client.py
import httpx
from typing import AsyncGenerator, Optional
import json
import re

class LettaClient:
    def __init__(self, base_url: str):
        self.base_url = base_url.rstrip("/")
        self.timeout = httpx.Timeout(60.0)

    async def list_agents(self) -> list[dict]:
        """Fetch all available agents"""
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.get(f"{self.base_url}/v1/agents")
            response.raise_for_status()
            return response.json()

    async def get_agent_memory(self, agent_id: str) -> dict:
        """Fetch agent memory blocks"""
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.get(f"{self.base_url}/v1/agents/{agent_id}/memory")
            response.raise_for_status()
            return response.json()

    async def stream_message(
        self,
        agent_id: str,
        message: str,
        system: Optional[str] = None
    ) -> AsyncGenerator[dict, None]:
        """Stream response from Letta agent via SSE"""
        prompt = f"{system}\n\n{message}" if system else message

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            async with client.stream(
                "POST",
                f"{self.base_url}/v1/agents/{agent_id}/messages/stream",
                json={"messages": [{"role": "user", "content": prompt}]},
                headers={"Content-Type": "application/json"}
            ) as response:
                async for line in response.aiter_lines():
                    if line.startswith("data: "):
                        data = line[6:]
                        if data == "[DONE]":
                            break
                        try:
                            payload = json.loads(data)
                            event = self._process_event(payload)
                            if event:
                                yield event
                        except json.JSONDecodeError:
                            continue

    def _process_event(self, payload: dict) -> Optional[dict]:
        """Process SSE payload and extract relevant events"""
        message_type = payload.get("message_type")

        if message_type == "tool_call_message":
            tool_call = payload.get("tool_call", {})
            return {
                "type": "tool_call",
                "tool_name": tool_call.get("name", ""),
                "arguments": tool_call.get("arguments", {})
            }

        if message_type == "assistant_message":
            content = payload.get("content", "")
            sanitized = self._sanitize_content(content)
            if sanitized:
                return {"type": "text", "content": sanitized}

        return None

    def _sanitize_content(self, content: str) -> str:
        """Remove tool metadata, UUIDs, etc from response"""
        # Port sanitization patterns from slackbot/ai/providers/letta_stream.py
        if not content or not isinstance(content, str):
            return ""

        # Remove tool call patterns
        patterns = [
            r"^[A-Z][a-z]+[A-Z][a-z]+.*$",  # PascalCase tool names
            r"^(successcall|failedcall)-.*$",  # Tool call IDs
            r"^message-.*$",  # Message IDs
            r"^step-.*$",  # Step IDs
        ]

        for pattern in patterns:
            if re.match(pattern, content.strip()):
                return ""

        return content.strip()

    # ----------------------------------------------------------------
    # Pattern 3: Archival Persistence
    # ----------------------------------------------------------------

    async def create_passage(
        self,
        agent_id: str,
        text: str,
        tags: list[str] = None
    ) -> dict:
        """
        Write passage to agent's archival memory.
        Fire-and-forget: caller should use asyncio.create_task().

        Args:
            agent_id: The agent whose archival to write to (typically main agent)
            text: The passage text to store
            tags: List of tags for filtering (e.g., ["session:2026-01-07", "agent:calendar"])
        """
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(
                f"{self.base_url}/v1/agents/{agent_id}/archival",
                json={"text": text, "tags": tags or []}
            )
            response.raise_for_status()
            return response.json()

    # ----------------------------------------------------------------
    # Pattern 4: Briefing Injection
    # ----------------------------------------------------------------

    async def list_passages(
        self,
        agent_id: str,
        tags: list[str] = None,
        limit: int = 5
    ) -> list[dict]:
        """
        Query passages from agent's archival memory for briefing injection.
        Called once before main agent, adds ~50-100ms latency.

        Args:
            agent_id: The agent whose archival to query
            tags: Filter tags (e.g., ["session:2026-01-07"])
            limit: Max passages to return
        """
        params = {"limit": limit}
        if tags:
            # Letta API may expect comma-separated or repeated params
            params["tags"] = ",".join(tags)

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.get(
                f"{self.base_url}/v1/agents/{agent_id}/archival",
                params=params
            )
            response.raise_for_status()
            data = response.json()
            return data.get("passages", data if isinstance(data, list) else [])

    def format_briefing(self, passages: list[dict]) -> str:
        """
        Format retrieved passages as briefing string for injection.
        Returns empty string if no passages (zero overhead).
        """
        if not passages:
            return ""

        lines = ["[Today's session briefing:]"]
        for p in passages:
            text = p.get("text", "")[:100]
            lines.append(f"  • {text}")
        return "\n".join(lines)
```
