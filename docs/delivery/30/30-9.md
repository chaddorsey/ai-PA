### 1. API Endpoints

```python
# app.py additions

@app.route('/api/agents')
def list_agents():
    """List all available Letta agents"""
    client = LettaClient(app.config['LETTA_BASE_URL'])
    agents = client.list_agents()

    return jsonify({
        "agents": [
            {
                "id": agent["id"],
                "name": agent.get("name", "Unknown"),
                "description": agent.get("description", ""),
                "capabilities": extract_capabilities(agent)
            }
            for agent in agents
        ]
    })

@app.route('/api/agents/<agent_id>/memory')
def get_agent_memory(agent_id):
    """Get memory blocks for an agent"""
    client = LettaClient(app.config['LETTA_BASE_URL'])
    memory = client.get_agent_memory(agent_id)

    return jsonify({
        "blocks": [
            {
                "label": block.get("label", ""),
                "value": block.get("value", ""),
                "limit": block.get("limit", 0)
            }
            for block in memory.get("memory", {}).get("blocks", [])
        ]
    })
```

### 2. Dashboard UI

```javascript
// dashboard.js
class Dashboard {
    constructor() {
        this.agentId = null;
        this.refreshInterval = null;
    }

    async loadAgents() {
        const response = await fetch('/api/agents');
        const data = await response.json();

        const select = document.getElementById('agent-select');
        select.innerHTML = data.agents.map(agent =>
            `<option value="${agent.id}">${agent.name}</option>`
        ).join('');

        // Select first agent
        if (data.agents.length > 0) {
            this.selectAgent(data.agents[0].id);
        }
    }

    selectAgent(agentId) {
        this.agentId = agentId;
        this.loadMemoryBlocks();
        this.startAutoRefresh();
    }

    async loadMemoryBlocks() {
        if (!this.agentId) return;

        const response = await fetch(`/api/agents/${this.agentId}/memory`);
        const data = await response.json();

        const container = document.getElementById('memory-blocks');
        container.innerHTML = data.blocks.map(block => `
            <div class="memory-block">
                <h4>${block.label}</h4>
                <pre>${escapeHtml(block.value)}</pre>
            </div>
        `).join('');
    }

    startAutoRefresh() {
        if (this.refreshInterval) {
            clearInterval(this.refreshInterval);
        }
        this.refreshInterval = setInterval(() => this.loadMemoryBlocks(), 30000);
    }
}
```
